Source — one script tag, one mount call

<!-- 153 KB gzip, CSS inlined -->
<script src="./damit-viewer.umd.js"></script>
<div id="host" style="width:100%; height:600px"></div>
<script>
  const base = './sample-data/hermione/';
  Promise.all([
    fetch(base + 'shape_damit_155.txt').then(r => r.text()),
    fetch(base + 'herm.lc').then(r => r.text()),
  ]).then(([shapeText, lcText]) => {
    const shape       = DAMITViewer.parseShapeTxt(shapeText);
    const lightCurves = DAMITViewer.parseLcTxt(lcText);
    const scattering  = { lambertWeight: 0.1 };

    // Auto-fit pole if the shape file and LCs come from different inversions.
    // Skip this if you already have the matching published pole + JD0.
    const seed  = { poleLambdaDeg: 293, poleBetaDeg: -34, periodHours: 5.55128, jd0: 2443032.5 };
    const facets = DAMITViewer.buildFacetGeometry(shape);
    const fit    = DAMITViewer.fitPoleAndPhase(facets, seed, scattering, lightCurves.slice(0, 5));

    DAMITViewer.mount(document.getElementById('host'), {
      name: '(121) Hermione',
      shape, lightCurves, scattering,
      spin: { ...seed, poleLambdaDeg: fit.poleLambdaDeg, poleBetaDeg: fit.poleBetaDeg, jd0: fit.jd0 },
      publishedSpin: seed,
      fitRms: fit.rms,
    }, {
      initialLightCurveIndex: 9,         // LC #10, high-amplitude apparition
      initialViewMode:        'earth',    // canonical 'view from Earth' frame
    });
  });
</script>

What's in the bundle

damit-viewer.umd.js is self-contained: Three.js, uPlot, and all styles are inlined, so a single <script> tag is enough — no separate stylesheet, no CDN dependency, no build step on the embedder side. The widget injects its own CSS at first mount(), scoped to .damit-viewer-host, with a :where()-based reset so a host page's !important rules on button/select don't bleed through.

The data you pass in is plain JS: parseShapeTxt and parseLcTxt accept the canonical DAMIT text formats, so a DAMIT model page can drop in the same text it's already serving from asteroid_models/files/.

Returned handle

const handle = DAMITViewer.mount(host, model, opts);
handle.setLightCurveByIndex(12);     // switch the displayed curve
handle.setLightCurveById(7);         // (alternative: by LC's `id` field)
handle.setJd(2452997.9);             // scrub to a specific epoch
handle.play(600);                     // 600× real time
handle.pause();
handle.setScatteringC(0.05);          // adjust Lambert weight live
handle.setViewMode('free');            // 'free' | 'earth'
handle.getModel();                    // returns a frozen snapshot
handle.dispose();                     // tears down on unmount
